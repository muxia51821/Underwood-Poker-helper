    lastCalc: {},
    init() {
      this.potInput = document.getElementById('potInput'); this.betInput = document.getElementById('betInput'); this.outsInput = document.getElementById('outsInput'); this.stackInput = document.getElementById('stackInput');

    },
    calc() {
      const pot = +this.potInput.value, bet = +this.betInput.value, outs = +this.outsInput.value, stack = +this.stackInput.value || 0;
      const isOne = document.querySelector('input[name="street"]:checked').value === 'one', combo = document.getElementById('comboCheck').checked, turnBet = combo ? +document.getElementById('turnBetInput').value : 0;
      const ratioDiv = document.getElementById('betRatio'), res = document.getElementById('oddsResult');
      if (isNaN(pot) || isNaN(bet) || pot <= 0 || bet <= 0) { ratioDiv.textContent = ''; res.innerHTML = '<div class="text-muted">请输入有效的底池和下注金额</div>'; return; }
      ratioDiv.textContent = `📊 对手下注了底池的 ${Utils.safeFixed(bet/pot*100, 0)}%`;
      if (stack > 0) { const spr = (stack / (pot + bet)).toFixed(1); document.getElementById('sprDisplay').innerHTML = !isNaN(spr) ? `💡 SPR: <span class="highlight">${spr}</span> (${stack}BB / ${pot+bet}BB)` : ''; } else { document.getElementById('sprDisplay').textContent = ''; }
      if (isNaN(outs) || outs < 0 || outs > 25 || (combo && (isNaN(turnBet) || turnBet <= 0))) { res.innerHTML = '<span class="text-lose">请输入有效的Outs (0-25) 及转牌下注</span>'; return; }
      let cost, totalPot, estWin, reqWin, ev;
      if (!combo) { cost = bet; totalPot = pot + bet * 2; reqWin = cost / totalPot; estWin = isOne ? Math.min(outs*2,100)/100 : Math.min(outs*4,100)/100; ev = estWin * totalPot - cost; }
      else { cost = bet + turnBet; totalPot = pot + bet*2 + turnBet*2; reqWin = cost / totalPot; estWin = Math.min(outs*4,100)/100; ev = estWin * totalPot - cost; }
      this.lastCalc = { pot, bet, callAmount: cost, totalPot, estWin, reqWin, combo, turnBet };
      const req1 = Math.ceil(reqWin*100/2), req2 = Math.ceil(reqWin*100/4), mdf = pot / (pot + bet), mdfPct = (mdf*100).toFixed(1);
      let html = `<b>🧾 计算</b><br>下注占底池 <span class="highlight">${Utils.safeFixed(bet/pot*100, 1)}%</span><br>`;
      if (combo) html += `转牌下注 ${turnBet} BB，总成本 <span class="highlight">${cost} BB</span><br>`;
      html += `需要胜率 ≥ <span class="highlight">${Utils.safeFixed(reqWin*100, 1)}%</span><br>`;
      html += `估算胜率 <span class="${estWin>=reqWin?'text-win':'text-lose'}">${Utils.safeFixed(estWin*100, 1)}%</span> (Outs×${combo?4:(isOne?2:4)})<br>`;
      html += `EV ≈ <span class="${ev>=0?'text-win':'text-lose'}">${Utils.safeFixed(ev, 2)} BB</span><br>`;
      html += `📌 平衡所需Outs: 一张 ≥ <span class="highlight">${req1}</span>, 两张 ≥ <span class="highlight">${req2}</span><br>`;
      html += `<br><b>🛡️ MDF</b><br>防守 <span class="highlight">${mdfPct}%</span> 的范围<br><br>`;
      html += estWin > reqWin ? '<span class="text-win">✅ +EV 跟注</span>' : (Math.abs(estWin-reqWin)<0.005 ? '<span class="highlight">⚖️ 边缘决策</span>' : '<span class="text-lose">❌ -EV 弃牌</span>');
      const betRatio = bet / pot;
      html += `<br><details style="margin-top:10px"><summary style="font-size:0.8em;color:#94a3b8;cursor:pointer">📊 常见下注尺度速查</summary><table class="cheatsheet"><thead><tr><th>尺度</th><th>所需胜率</th><th>MDF(防守)</th><th>诈唬需弃牌率</th></tr></thead><tbody>`;
      [{label:'1/4 pot',r:0.25},{label:'1/3 pot',r:1/3},{label:'1/2 pot',r:0.5},{label:'2/3 pot',r:2/3},{label:'3/4 pot',r:0.75},{label:'满池',r:1},{label:'1.25x',r:1.25},{label:'1.5x',r:1.5}].forEach(s => { const req = Utils.safeFixed(s.r/(1+2*s.r)*100, 0), mdfV = Utils.safeFixed(1/(1+s.r)*100, 0), blf = Utils.safeFixed(s.r/(1+s.r)*100, 0); html += `<tr class="${Math.abs(betRatio-s.r)<0.01?'is-current':''}"><td>${s.label}</td><td>${req}%</td><td>${mdfV}%</td><td>${blf}%</td></tr>`; });
      html += `</tbody></table></details>`;
      html += `<details id="impliedDetails"><summary style="font-size:0.8em;color:#94a3b8;cursor:pointer">💡 隐含赔率估算</summary><label class="field" style="margin-top:8px">预计听中后可额外赢取 (BB) <input type="number" id="impliedInput" class="input" value="0" step="1" min="0"></label><div id="impliedResult" class="text-muted" style="margin-top:8px">输入额外赢取以计算隐含EV</div></details>`;
      res.innerHTML = html;
      this.calcImplied();
    },
    calcImplied() { const implied = parseFloat(document.getElementById('impliedInput')?.value) || 0; const { callAmount, totalPot, estWin } = this.lastCalc; const resultDiv = document.getElementById('impliedResult'); if (implied > 0) { const impliedEv = estWin * (totalPot + implied) - callAmount; const impliedReqWin = callAmount / (totalPot + implied); resultDiv.innerHTML = `额外赢取 <span class="highlight">${implied} BB</span><br>隐含EV ≈ <span class="${impliedEv>=0?'text-win':'text-lose'}">${Utils.safeFixed(impliedEv, 2)} BB</span><br>所需胜率 ≈ <span class="highlight">${Utils.safeFixed(impliedReqWin*100, 1)}%</span>`; } else { resultDiv.innerHTML = '输入额外赢取以计算隐含EV'; } },
